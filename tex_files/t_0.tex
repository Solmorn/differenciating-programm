\begin{task}{2}
    Верны ли следующие утверждения:
    \begin{enumerate}
        \item $ \displaystyle{\frac{n}{\log n} = \Omega\!\left(\frac{n}{2}\right)} $;

        \item $ \displaystyle{n^{\log n} = O\!\left(1.01^n\right)} $;

        \item $ \displaystyle{\sqrt{n}^{\sqrt{n}} = O\!\left(\left(\log n\right)^n\right)} $;

        \item $ \displaystyle{n^n = O\!\left(n!\right)} $.
    \end{enumerate}
\end{task}

\begin{solution}
\begin{enumerate}
    \item $\frac{n}{\log n} = \Omega\!\left(\frac{n}{2}\right)$
    при больших $n$:
    \begin{equation*}
        \begin{split}  
            \frac{n}{\log n} \geqslant C\cdot\frac{n}{2}\\
            \frac{2}{C} \geqslant \log n\\
        \end{split}
    \end{equation*} что неверно
    \item $n^{\log n} = O\!\left(1.01^n\right)$
    при больших $n$:
    \begin{equation*}
        \begin{split}  
            n^{\log n} \leqslant C\cdot1.01^n\\
            \log_a n \leqslant \sqrt{\log_a C + n\log_a1.01}\\
        \end{split}
    \end{equation*} что верно
    \item $\sqrt{n}^{\sqrt{n}} = O\!\left(\left(\log n\right)^n\right)$
    при больших $n$:
    \begin{equation*}
        \begin{split}  
            \sqrt{n}^{\sqrt{n}} \leqslant C\cdot\left(\log n\right)^n\\
            \sqrt n \log_a \sqrt n \leqslant \log_a C + \sqrt n \log_a{(\log_a n)^\sqrt n}\\
            \sqrt n \log_a{\frac{\sqrt n}{(\log_a n)^\sqrt n}} \leqslant \log_a C\\
        \end{split}
    \end{equation*} что верно, в левой части неравенства будет число стремящееся к $-\infty$
    \item $n^n = O\!\left(n!\right)$
    при больших $n$:
    \begin{equation*}
        \begin{split}  
            n^n \leqslant C\cdot n!\\
            \frac{n^n}{n!}\leqslant C\\
        \end{split}
    \end{equation*} что неверно
\end{enumerate}
\end{solution}
